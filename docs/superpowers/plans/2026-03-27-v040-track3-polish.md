```rust
    }));
}
```

And in `RateLimitBanner.tsx`:

```tsx
useEffect(() => {
  if (!isTauriContext()) return;
  const unlisten = listen<{ provider: string; retry_after_ms: number }>('rate-limit-hit', (event) => {
    setVisible(true);
    setProvider(event.payload.provider);
    setTimeout(() => setVisible(false), event.payload.retry_after_ms + 2000);
  });
  return () => { unlisten.then(f => f()); };
}, []);
```
