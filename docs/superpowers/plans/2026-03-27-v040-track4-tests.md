```rust
#[cfg(test)]
mod tests {
    use super::*;

    /// Build a price series from a slice of close prices.
    fn prices(closes: &[f64]) -> Vec<HistoricalPrice> {
        closes
            .iter()
            .enumerate()
            .map(|(i, &c)| HistoricalPrice {
                date: format!("2024-01-{:02}", i + 1),
                close: c,
            })
            .collect()
    }

    /// A flat price series — zero volatility, zero return.
    fn flat_prices(n: usize, price: f64) -> Vec<HistoricalPrice> {
        prices(&vec![price; n])
    }

    /// A linearly rising series — known return, low volatility.
    fn rising_prices(n: usize, start: f64, step: f64) -> Vec<HistoricalPrice> {
        let closes: Vec<f64> = (0..n).map(|i| start + step * i as f64).collect();
        prices(&closes)
    }

    #[test]
    fn test_insufficient_data_returns_fallback() {
        let p = prices(&[100.0, 101.0]); // Only 2 points, below MIN_DATA_POINTS (14)
        let m = QuantAnalyzer::calculate_metrics("AAPL", &p);
        assert_eq!(m.symbol, "AAPL");
        // Should return zeros/neutral values, not NaN or panic
        assert!(!m.sharpe_ratio.is_nan());
        assert!(!m.volatility.is_nan());
        assert!(!m.annualized_return.is_nan());
    }

    #[test]
    fn test_flat_prices_sharpe_is_near_zero() {
        // Flat prices → zero daily returns → zero excess return → Sharpe ≈ 0
        let p = flat_prices(30, 100.0);
        let m = QuantAnalyzer::calculate_metrics("FLAT", &p);
        assert!(!m.sharpe_ratio.is_nan());
        // Sharpe should be close to 0 (or negative due to risk-free rate)
        assert!(m.sharpe_ratio.abs() < 5.0, "Flat price Sharpe should be near zero, got {}", m.sharpe_ratio);
        assert_eq!(m.max_drawdown, 0.0, "Flat price max drawdown should be 0");
    }

    #[test]
    fn test_rising_prices_positive_return() {
        // Rising prices → positive annualized return
        let p = rising_prices(252, 100.0, 0.5); // $100 → $225.5 over 252 days
        let m = QuantAnalyzer::calculate_metrics("RISING", &p);
        assert!(m.annualized_return > 0.0, "Rising prices should have positive return, got {}", m.annualized_return);
        assert_eq!(m.max_drawdown, 0.0, "Monotonically rising prices have no drawdown");
    }

    #[test]
    fn test_drawdown_on_v_shaped_series() {
        // Price falls 50% then recovers — max drawdown should be ~50%
        let mut closes = vec![100.0; 10];
        closes.extend(vec![50.0; 10]); // 50% drop
        closes.extend(vec![100.0; 10]); // full recovery
        let p = prices(&closes);
        let m = QuantAnalyzer::calculate_metrics("VSHAPED", &p);
        // Max drawdown should be close to 0.5 (50%)
        assert!(m.max_drawdown >= 0.45 && m.max_drawdown <= 0.55,
            "V-shape max drawdown should be ~50%, got {}", m.max_drawdown);
    }

    #[test]
    fn test_calculate_metrics_no_nan_or_inf() {
        // Any valid price series should produce finite metrics
        let p = rising_prices(252, 50.0, 0.2);
        let m = QuantAnalyzer::calculate_metrics("AAPL", &p);
        assert!(m.sharpe_ratio.is_finite(), "Sharpe must be finite");
        assert!(m.annualized_return.is_finite(), "Return must be finite");
        assert!(m.volatility.is_finite(), "Volatility must be finite");
        assert!(m.max_drawdown.is_finite(), "Drawdown must be finite");
        assert!(m.rsi.is_finite(), "RSI must be finite");
    }

    #[test]
    fn test_rsi_bounded_0_to_100() {
        let p = rising_prices(30, 100.0, 1.0);
        let m = QuantAnalyzer::calculate_metrics("RSI_TEST", &p);
        assert!(m.rsi >= 0.0 && m.rsi <= 100.0,
            "RSI must be in [0, 100], got {}", m.rsi);
    }

    #[test]
    fn test_signal_is_valid_string() {
        let p = rising_prices(30, 100.0, 0.5);
        let m = QuantAnalyzer::calculate_metrics("SIG_TEST", &p);
        let valid_signals = ["BUY", "STRONG BUY", "SELL", "STRONG SELL", "NEUTRAL", "HOLD"];
        assert!(valid_signals.iter().any(|&s| m.signal.contains(s)),
            "Signal must be a valid value, got '{}'", m.signal);
    }
}
```
